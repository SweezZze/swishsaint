# Coutures : les SEULS endroits où le fork modifie un fichier officiel.
#
# Format :   <id>   <fichier>   # explication
# Dans le fichier, chaque ligne modifiée porte le marqueur   // FORK(<id>)
# Une couture = 1 à 3 lignes qui appellent du code rangé dans Sources/Vorssaint/Fork/.
# fork/check-seams.sh refuse toute modification d'un fichier officiel qui n'est pas listée ici.
#
bootstrap      Sources/Vorssaint/App/AppDelegate.swift                        # démarre le code du fork au lancement
update         Sources/Vorssaint/Services/Update/UpdateService.swift          # coupe l'updater officiel, branche celui du fork
backup         Sources/Vorssaint/Services/SettingsBackup.swift                # les réglages du fork partent dans la sauvegarde et en reviennent
feedback       Sources/Vorssaint/Services/Feedback/FeedbackService.swift      # n'envoie pas nos rapports de bug au dev officiel
settings       Sources/Vorssaint/UI/Settings/FeatureVisibilitySupport.swift   # pages du fork : déclare « Extras » et les pages des features (.fork(id))
settings2      Sources/Vorssaint/UI/Settings/SettingsView.swift               # pages du fork : affiche « Extras » et chaque page de feature
settings3      Sources/Vorssaint/UI/Settings/SettingsDirectory.swift          # page « Extras » : sa ligne dans la barre latérale et la recherche
settings4      Sources/Vorssaint/UI/Settings/SettingsDirectory.swift          # pages des features du fork : leur ligne dans la barre latérale et la recherche
window-layout  Sources/Vorssaint/Services/WindowLayout/WindowLayoutService.swift  # gestes du trackpad : suivent chaque synchro et chaque arrêt de Disposition des fenêtres
dictation-history  Sources/Vorssaint/Services/Clipboard/ClipboardHistoryService.swift  # dictée : chaque transcription entre dans l'historique du presse-papiers, avec l'app comme source (promote est privé)
notch-dictations   Sources/Vorssaint/UI/Notch/NotchView.swift                  # dictée : onglet « Dictées » de la page Presse-papiers de l'île (sélecteur d'onglets dans la rangée du haut)
notch-dictation    Sources/Vorssaint/Services/Notch/NotchSupport.swift         # dictée : bouton « Dictées » des contrôles de l'île (cas et disponibilité injectée) — ne nomme aucun type du fork (TEST_SOURCES)
notch-dictation2   Sources/Vorssaint/Services/Notch/NotchSupport.swift         # dictée : bouton « Dictées » (symbole, réglage requis, disponibilité) — ne nomme aucun type du fork (TEST_SOURCES)
notch-dictation3   Sources/Vorssaint/UI/Notch/NotchControlsView.swift          # dictée : tuile et titre du bouton « Dictées » dans les contrôles de l'île
notch-dictation4   Sources/Vorssaint/Services/Notch/NotchService.swift         # dictée : bouton rapide « Dictées » ; clic sur la notice d'une dictée → onglet Dictées
notch-dictation5   Sources/Vorssaint/UI/Settings/NotchSettings.swift           # dictée : raison et lien du bouton « Dictées » tant que la Dictée est coupée (sinon la règle officielle de la page Presse-papiers)
notch-dictation6   Sources/Vorssaint/Services/Clipboard/ClipboardHistoryService.swift  # dictée : raccourci et menu de l'historique du presse-papiers sur l'onglet Dictées → reviennent à l'onglet Presse-papiers (showClipboard est extrait par les tests officiels)
notch-dictation7   Sources/Vorssaint/Services/Notch/NotchService.swift         # dictée : Explorer et les boutons qui choisissent la page Presse-papiers montrent son onglet Presse-papiers
